1. Architecture Overview

Guardian operates as a sidecar to Aperture, Tailscale's LLM proxy. It has two faces: a pre-request hook that Aperture calls before forwarding requests to LLM providers, and a service proxy that agents call instead of external APIs.

Guardian sits alongside Aperture: the hook intercepts LLM requests, the proxy mediates service data.

3. Entry Point — main.go

The entry point parses flags, loads config, and wires up two HTTP routes: the Aperture hook and the service proxy.

Go
func main() {
    addr := flag.String("addr", ":8080", "listen address")
    baseURL := flag.String("base-url", "", "...")
    apiKey := flag.String("api-key", "", "...")
    flag.Parse()

    cfg, err := LoadConfig(*configPath)
    if err != nil {
        slog.Error("failed to load config",
            "error", err)
        os.Exit(1)
    }

    srv := &Server{
        apiKey:  *apiKey,
        baseURL: *baseURL,
        config:  cfg,
        cache:   NewCache(),
    }

    mux := http.NewServeMux()
    mux.HandleFunc(
        "POST /v1/pre-request",
        srv.AuthMiddleware(
            srv.HandlePreRequest))
    mux.HandleFunc(
        "/v1/services/",
        srv.AuthMiddleware(
            srv.HandleServiceProxy))

    http.ListenAndServe(*addr, mux)
}

4. Configuration — config.go

Guardian's config defines named services. Each service has an action (inject or block), an upstream endpoint, and optional caching/auth settings.

4.1 Data Structures

Go — structs with JSON tags
type Config struct {
    mu       sync.RWMutex
    Services []Service `json:"services"`
    path     string
}

type Service struct {
    ID       string            `json:"id"`
    Name     string            `json:"name"`
    Aliases  []string          `json:"aliases,omitempty"`
    Action   string            `json:"action"`
    Endpoint string            `json:"endpoint,omitempty"`
    Headers  map[string]string `json:"headers,omitempty"`
    CacheTTL Duration          `json:"cache_ttl"`
}

// Custom JSON marshalling for time.Duration
type Duration struct{ time.Duration }

func (d *Duration) UnmarshalJSON(
    b []byte,
) error {
    var s string
    json.Unmarshal(b, &s)
    parsed, err := time.ParseDuration(s)
    d.Duration = parsed
    return err
}
Python — dataclasses
from dataclasses import dataclass, field
from datetime import timedelta

@dataclass
class Service:
    id: str
    name: str
    action: str  # "inject" or "block"
    aliases: list[str] = field(
        default_factory=list)
    endpoint: str = ""
    headers: dict[str, str] = field(
        default_factory=dict)

@dataclass
class Config:
    services: list[Service]

def load_config(path: str) -> Config:
    with open(path) as f:
        data = json.load(f)
    return Config(
        services=[
            Service(**s)
            for s in data["services"]
        ]
    )

Go note: Struct tags like `json:"id"` tell Go's JSON encoder how to map field names. Python's dataclass + json.load achieves the same with less ceremony. Go requires a custom Duration type because time.Duration doesn't natively marshal to JSON as "10m".

4.2 Service Matching

Finding which services are referenced in a user message — case-insensitive substring matching on the service name and aliases:

Go
func (c *Config) FindMatchingServices(
    userMessage string,
) []Service {
    c.mu.RLock()
    defer c.mu.RUnlock()

    lower := strings.ToLower(userMessage)
    var matched []Service
    for _, svc := range c.Services {
        if matchesService(svc, lower) {
            matched = append(matched, svc)
        }
    }
    return matched
}

func matchesService(
    svc Service, lowerMsg string,
) bool {
    if strings.Contains(
        lowerMsg,
        strings.ToLower(svc.Name)) {
        return true
    }
    for _, alias := range svc.Aliases {
        if strings.Contains(
            lowerMsg,
            strings.ToLower(alias)) {
            return true
        }
    }
    return false
}
Python
def find_matching_services(
    config: Config,
    user_message: str,
) -> list[Service]:
    lower = user_message.lower()
    return [
        svc for svc in config.services
        if matches_service(svc, lower)
    ]

def matches_service(
    svc: Service, lower_msg: str,
) -> bool:
    if svc.name.lower() in lower_msg:
        return True
    return any(
        alias.lower() in lower_msg
        for alias in svc.aliases
    )

Go note: c.mu.RLock() / defer c.mu.RUnlock() acquires a read lock for the duration of the function. This allows concurrent reads but blocks writes. Python doesn't need this because of the GIL (though a real service would use asyncio.Lock).

5.2 The Core Decision Logic

Go
func (s *Server) HandlePreRequest(
    w http.ResponseWriter,
    r *http.Request,
) {
    // 1. Parse payload
    var payload PreRequestPayload
    json.Unmarshal(body, &payload)

    // 2. Match services
    matched := s.config.FindMatchingServices(
        payload.UserMessage)

    if len(matched) == 0 {
        writeGuardrailResponse(w,
            GuardrailResponse{
                Action: "allow"})
        return
    }

    // 3. Block if any service is blocked
    for _, svc := range matched {
        if svc.Action == "block" {
            writeGuardrailResponse(w,
                GuardrailResponse{
                    Action: "block",
                    Message: fmt.Sprintf(
                        "blocked: %s",
                        svc.Name)})
            return
        }
    }

    // 4. Build proxy instructions
    var instructions []string
    for _, svc := range matched {
        if svc.Action != "inject" {
            continue
        }
        proxyURL := fmt.Sprintf(
            "%s/v1/services/%s",
            s.baseURL, svc.ID)
        instructions = append(instructions,
            fmt.Sprintf(
                "use %s for %s",
                proxyURL, svc.Name))
    }

    // 5. Inject into LLM request
    modified := injectSystemMessage(
        payload.RequestBody,
        strings.Join(instructions, "\n"))
    writeGuardrailResponse(w,
        GuardrailResponse{
            Action: "modify",
            Body:   modified})
}
Python
async def handle_pre_request(
    payload: dict,
    config: Config,
    cache: Cache,
) -> dict:
    req = PreRequestPayload(**payload)

    # 2. Match services
    matched = find_matching_services(
        config, req.user_message)

    if not matched:
        return {"action": "allow"}

    # 3. Block if any service is blocked
    for svc in matched:
        if svc.action == "block":
            return {
                "action": "block",
                "message": f"blocked: {svc.name}",
            }

    # 4. Build proxy instructions
    instructions = []
    for svc in matched:
        if svc.action != "inject":
            continue
        proxy_url = (
            f"{BASE_URL}/v1/services/{svc.id}")
        instructions.append(
            f"use {proxy_url} for {svc.name}")

    # 5. Inject into LLM request
    modified = inject_system_message(
        req.request_body,
        "\n".join(instructions))
    return {
        "action": "modify",
        "body": modified,
    }

5.3 Injecting a System Message

This function prepends a system message to the LLM request's message array, preserving all other fields (model, temperature, etc.):

Go
func injectSystemMessage(
    original []byte,
    content string,
) (json.RawMessage, error) {
    var req struct {
        Messages []json.RawMessage
    }
    json.Unmarshal(original, &req)

    sysMsg, _ := json.Marshal(
        map[string]string{
            "role":    "system",
            "content": content,
        })

    // Prepend to messages array
    req.Messages = append(
        []json.RawMessage{sysMsg},
        req.Messages...)

    // Rebuild with all original fields
    var full map[string]json.RawMessage
    json.Unmarshal(original, &full)
    full["messages"], _ =
        json.Marshal(req.Messages)
    return json.Marshal(full)
}
Python
def inject_system_message(
    original: dict,
    content: str,
) -> dict:
    # Deep copy to avoid mutating original
    modified = json.loads(
        json.dumps(original))

    sys_msg = {
        "role": "system",
        "content": content,
    }

    # Prepend to messages array
    modified["messages"] = (
        [sys_msg] + modified["messages"])

    return modified

Go note: json.RawMessage is a byte slice that holds unparsed JSON. It lets us delay parsing — we can read the messages array without fully parsing every message object. Python doesn't need this because dict is already JSON-native.

5.4 Auth Middleware

Go — closure-based middleware
func (s *Server) AuthMiddleware(
    next http.HandlerFunc,
) http.HandlerFunc {
    // Returns a new handler that wraps `next`
    return func(
        w http.ResponseWriter,
        r *http.Request,
    ) {
        auth := r.Header.Get(
            "Authorization")
        if auth != "Bearer "+s.apiKey {
            if r.Header.Get(
                "X-Api-Key") != s.apiKey {
                http.Error(w,
                    `{"error":"unauthorized"}`,
                    401)
                return
            }
        }
        next(w, r) // call the real handler
    }
}
Python — decorator or dependency

6. Service Proxy — proxy.go

The proxy is the second half of the architecture. When the LLM agent (following the injected instructions) calls /v1/services/spotify/playlists, Guardian forwards to the real Spotify API with proper auth headers, and caches the response.

Proxy request flow: cache-first for GET, always forward for POST.

Go — proxy handler
func (s *Server) HandleServiceProxy(
    w http.ResponseWriter,
    r *http.Request,
) {
    // Extract service ID from URL path
    parts := strings.SplitN(
        strings.TrimPrefix(
            r.URL.Path, "/v1/services/"),
        "/", 2)
    serviceID := parts[0]

    svc := s.config.GetService(serviceID)
    if svc == nil {
        writeJSONError(w, 404, "not found")
        return
    }

    // Cache check for GET requests
    cacheKey := serviceID + ":" +
        r.URL.RequestURI()
    if r.Method == "GET" {
        if cached, ok :=
            s.cache.Get(cacheKey); ok {
            w.Header().Set(
                "X-Guardian-Cache", "hit")
            io.WriteString(w, cached)
            return
        }
    }

    // Build upstream request
    upstreamURL := svc.Endpoint + remainingPath
    req, _ := http.NewRequestWithContext(
        r.Context(), r.Method,
        upstreamURL, r.Body)

    // Apply service auth headers
    for k, v := range svc.Headers {
        req.Header.Set(k, v)
    }

    resp, _ := http.DefaultClient.Do(req)
    data, _ := io.ReadAll(resp.Body)

    // Cache GET responses
    if r.Method == "GET" {
        s.cache.Set(cacheKey,
            string(data), svc.CacheTTL.Duration)
    }

    w.Write(data)
}
Python — proxy handler
import httpx

async def handle_proxy(
    service_id: str,
    path: str,
    request: Request,
    config: Config,
    cache: Cache,
) -> Response:
    svc = config.get_service(service_id)
    if not svc:
        return JSONResponse(
            {"error": "not found"},
            status_code=404)

    # Cache check for GET
    cache_key = f"{service_id}:{request.url}"
    if request.method == "GET":
        cached = cache.get(cache_key)
        if cached:
            return JSONResponse(
                cached,
                headers={"X-Guardian-Cache":
                         "hit"})

    # Forward to upstream
    upstream = f"{svc.endpoint}/{path}"
    async with httpx.AsyncClient() as c:
        resp = await c.request(
            method=request.method,
            url=upstream,
            headers={**request.headers,
                     **svc.headers},
            content=await request.body())

    # Cache GET responses
    if request.method == "GET":
        cache.set(cache_key,
               resp.json(),
               svc.cache_ttl)

    return JSONResponse(
        resp.json(),
        status_code=resp.status_code,
        headers={"X-Guardian-Cache":
                 "miss"})

7. Cache — cache.go

A simple in-memory TTL cache. Entries expire automatically on read. Protected by a read-write mutex for concurrent access.

Go — mutex-protected map
type Cache struct {
    mu    sync.RWMutex
    items map[string]cacheEntry
}

type cacheEntry struct {
    value   string
    expires time.Time
}

func (c *Cache) Get(key string) (string, bool) {
    c.mu.RLock()
    defer c.mu.RUnlock()
    entry, ok := c.items[key]
    if !ok || time.Now().After(entry.expires) {
        return "", false
    }
    return entry.value, true
}

func (c *Cache) Set(
    key, value string, ttl time.Duration,
) {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.items[key] = cacheEntry{
        value:   value,
        expires: time.Now().Add(ttl),
    }
}
Python — dict with expiry
import time
from threading import Lock

class Cache:
    def __init__(self):
        self._items: dict[str, tuple] = {}
        self._lock = Lock()

    def get(self, key: str) -> str | None:
        with self._lock:
            entry = self._items.get(key)
            if not entry:
                return None
            value, expires = entry
            if time.time() > expires:
                del self._items[key]
                return None
            return value

    def set(self, key: str,
            value: str,
            ttl: timedelta):
        with self._lock:
            self._items[key] = (
                value,
                time.time() + ttl.total_seconds()
            )

Go note: sync.RWMutex allows multiple concurrent readers (via RLock) but exclusive writers (via Lock). The defer keyword ensures the lock is released when the function returns, even on error — similar to Python's with statement.

Error Handling

Go — explicit error returns

data, err := os.ReadFile(path)

}

Python — exceptions

Pointers and References

Go — explicit pointers
// & takes address, * dereferences
srv := &Server{apiKey: "key"}

// Methods on pointer receivers
func (s *Server) Handle(...) { }

// nil check before use
if svc == nil { return }
Python — everything is a reference
# Python objects are always references
srv = Server(api_key="key")

# Methods are always on references
def handle(self, ...): pass

# None check before use
if svc is None: return
