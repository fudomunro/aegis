1. Architecture Overview

Guardian has evolved from a simple proxy into a connector framework. The core idea: each music service (Spotify, YouTube Music, Amazon Music) gets its own connector that understands the service's API, transforms responses for LLM consumption, and declares what operations it supports.

Each service gets its own connector. Guardian's proxy delegates to the right connector based on the service ID in the URL.

2. The ServiceConnector Interface

This is the central abstraction. Every service implements these three methods:

Go — interface definition
type ServiceConnector interface {
    // What data types and operations this connector supports.
    Capabilities() ServiceCapabilities

    // System message injected into LLM requests.
    SystemInstructions(
        baseURL string,
    ) string

    // Handle a proxied request.
    HandleProxy(
        w http.ResponseWriter,
        r *http.Request,
        remainingPath string,
    )
}
Python — equivalent ABC
from abc import ABC, abstractmethod

class ServiceConnector(ABC):
    @abstractmethod
    def capabilities(self) -> ServiceCapabilities:
        """What this connector can do."""

    @abstractmethod
    def system_instructions(
        self, base_url: str,
    ) -> str:
        """Instructions for the LLM."""

    @abstractmethod
    async def handle_proxy(
        self, request: Request,
        path: str,
    ) -> Response:
        """Handle a proxied request."""

Go note: Go interfaces are satisfied implicitly — any struct with these three methods automatically implements ServiceConnector, no implements keyword needed. Python requires explicit inheritance from ABC.

3. Capabilities & Operations

Each connector declares exactly what it can do, with structured operation definitions that include method, path, parameters, and response format. This metadata is exposed at GET /v1/services/{id}/capabilities.

types in handler.go
type ServiceCapabilities struct {
    DataTypes []DataTypeCapability `json:"data_types"`
}

type DataTypeCapability struct {
    Name        string      `json:"name"`        // "tracks"
    Description string      `json:"description"`
    Operations  []Operation `json:"operations"`
}

type Operation struct {
    Name        string       `json:"name"`       // "read_saved"
    Description string       `json:"description"`
    Method      string       `json:"method"`     // "GET"
    Path        string       `json:"path"`       // "/me/tracks"
    Parameters  []Parameter  `json:"parameters,omitempty"`
    Response    ResponseSpec `json:"response"`
}

type Parameter struct {
    Name        string `json:"name"`
    In          string `json:"in"`          // "query" or "path"
    Type        string `json:"type"`        // "integer", "string"
    Description string `json:"description"`
    Default     any    `json:"default,omitempty"`
    Required    bool   `json:"required,omitempty"`
}

type ResponseSpec struct {
    Description string   `json:"description"`
    Fields      []string `json:"fields,omitempty"`
}

5. OAuth & Token Storage

Guardian manages OAuth tokens centrally so agents never see credentials. Tokens are encrypted at rest with AES-256-GCM.

OAuth flow: user authorizes via browser, Guardian exchanges code for tokens, encrypts and stores them.

tokens.go — encrypted storage
type TokenStore struct {
    mu     sync.RWMutex
    path   string
    salt   []byte
    key    []byte
    Tokens map[string]*OAuthToken
}

// AES-256-GCM with PBKDF2-derived key
func deriveKey(
    passphrase string, salt []byte,
) []byte {
    return pbkdf2.Key(
        []byte(passphrase), salt,
        100_000,  // iterations
        32,       // key length
        sha256.New,
    )
}

// On-disk format:
// { version: 1, salt: "base64...",
//   tokens: { "spotify": "encrypted..." } }
Python equivalent
from cryptography.fernet import Fernet
from hashlib import pbkdf2_hmac
import base64, json

class TokenStore:
    def __init__(self, path, passphrase):
        self.path = path
        self.key = derive_key(passphrase)
        self.tokens = {}

    def save(self, service_id, token):
        self.tokens[service_id] = token
        self._persist()

    def _persist(self):
        # Encrypt each token with AES-GCM
        encrypted = {
            sid: self._encrypt(
                json.dumps(t).encode()
            )
            for sid, t in self.tokens.items()
        }
        with open(self.path, "w") as f:
            json.dump({
                "version": 1,
                "tokens": encrypted,
            }, f)

7. Response Transformation

Each connector transforms the upstream API's verbose JSON into compact, LLM-friendly formats. Here's how Spotify's saved tracks endpoint is transformed:

Raw Spotify Response
{
  "items": [{
    "added_at": "2024-01-15T...",
    "track": {
      "album": {
        "name": "Kind of Blue",
        "images": [{...}, {...}],
        "external_urls": {...},
        "release_date": "1959-08-17",
        "total_tracks": 5
      },
      "artists": [{
        "name": "Miles Davis",
        "external_urls": {...},
        "images": [{...}],
        "followers": {...}
      }],
      "name": "So What",
      "id": "abc123",
      "uri": "spotify:track:abc123",
      "duration_ms": 562000,
      "popularity": 78
    }
  }],
  "next": "https://api.spotify.com/...",
  "offset": 0,
  "limit": 20,
  "total": 347
}
Transformed (LLM-friendly)
{
  "tracks": [{
    "name": "So What",
    "artists": ["Miles Davis"],
    "album": "Kind of Blue",
    "id": "abc123",
    "uri": "spotify:track:abc123"
  }],
  "total": 347,
  "total_fetched": 347
}

What's stripped: added_at, images, external_urls, followers, release_date, duration_ms, popularity, pagination metadata (next, offset, limit). All pages are auto-fetched and combined.

sync.RWMutex for concurrent access

Go — explicit locking
type TokenStore struct {
    mu     sync.RWMutex
    Tokens map[string]*OAuthToken
}

func (ts *TokenStore) Get(
    id string,
) *OAuthToken {
    ts.mu.RLock()         // read lock
    defer ts.mu.RUnlock() // auto-unlock
    return ts.Tokens[id]
}

func (ts *TokenStore) Save(
    id string, tok *OAuthToken,
) error {
    ts.mu.Lock()        // exclusive lock
    defer ts.mu.Unlock()
    ts.Tokens[id] = tok
    return ts.persist()
}
Python — threading or asyncio
from threading import Lock

class TokenStore:
    def __init__(self):
        self._lock = Lock()
        self.tokens = {}

    def get(self, sid):
        with self._lock:  # context mgr
            return self.tokens.get(sid)

    def save(self, sid, tok):
        with self._lock:
            self.tokens[sid] = tok
            self._persist()

# Go: RWMutex allows concurrent reads
# Python: Lock is always exclusive
# For async: use asyncio.Lock()
